SELECT mem_id AS '회원 아이디', SUM(price * amount) AS '총 구매 금액' FROM buy; -- buy 테이블에서 회원 아이디와 총 구매 금액 조회
SELECT mem_id AS '회원 아이디', SUM(price * amount) AS '구매금액' FROM buy GROUP BY mem_id; -- buy 테이블에서 회원아이디 별로 회원 아이디와 총 구매 금액 조회
SELECT mem_id AS '회원 아이디', SUM(price * amount) AS '구매금액' FROM buy GROUP BY mem_id ORDER BY SUM(price * amount) DESC; -- buy 테이블에서 회원아이디 별로 회원 아이디와 총 구매 금액 을 조회하는데 총 구매 금액 내림차순으로 조회
SELECT mem_id AS '회원 아이디', SUM(price * amount) AS '구매금액' FROM buy GROUP BY mem_id ORDER BY SUM(price * amount) DESC limit 2; -- buy 테이블에서 회원아이디 별로 회원 아이디와 총 구매 금액 을 조회하는데 총 구매 금액 내림차순으로 조회하는데 상위 2명만 조회
SELECT mem_id AS '회원 아이디', SUM(price * amount) AS '구매금액' FROM buy GROUP BY mem_id having SUM(price * amount) >= 1000 ORDER BY SUM(price * amount) DESC; -- buy 테이블에서 회원아이디 별로 회원 아이디와 총 구매 금액 을 조회하는데 총 구매 금액 내림차순으로 조회하는데 총 구매금액이 1000이상인 회원을 총 구매 금액 내림차순으로 조회
